package monopoly

import "time"

// ── Board ───────────────────────────────────────────────────────────────────

type TileType string

const (
	TileStart    TileType = "start"
	TileProperty TileType = "property"
	TileRail     TileType = "rail"
	TileUtility  TileType = "utility"
	TileEvent    TileType = "event"   // Chance
	TileMystery  TileType = "mystery" // Community Chest
	TileTax      TileType = "tax"
	TileJail     TileType = "jail"
	TileParking  TileType = "parking"
	TileGoToJail TileType = "gotojail"
)

type Tile struct {
	ID    int
	Name  string
	Type  TileType
	Group string // color group for properties
	Price int
}

// Board is the canonical 40-tile layout. Keep in sync with
// Frontend/lib/monopoly.ts MONO_BOARD (same ids, names, prices, groups).
var Board = [40]Tile{ /* 0 GO … 39 Boardwalk — fill from the frontend board */ }

// ── State ───────────────────────────────────────────────────────────────────

type Team struct {
	ID        int64
	Name      string
	Color     string
	Cash      int
	Position  int
	InJail    bool
	Bankrupt  bool
	Props     map[int]bool // tile id -> owned
	Houses    map[int]int  // tile id -> house count
}

func (t *Team) NetWorth() int {
	nw := t.Cash
	for id := range t.Props {
		nw += Board[id].Price
		nw += t.Houses[id] * 50
	}
	return nw
}

type State struct {
	Seed      int64
	Round     int
	Phase     string
	TurnSeat  int
	Teams     []*Team
	Seq       int64
	Deadline  time.Time
	Winner    *int64
}

// ── Agent actions (server-validated) ────────────────────────────────────────

type ActionKind string

const (
	RollDice    ActionKind = "ROLL_DICE"
	BuyProperty ActionKind = "BUY_PROPERTY"
	Build       ActionKind = "BUILD"
	Sell        ActionKind = "SELL"
	Mortgage    ActionKind = "MORTGAGE"
	Unmortgage  ActionKind = "UNMORTGAGE"
	Trade       ActionKind = "TRADE"
	AcceptTrade ActionKind = "ACCEPT_TRADE"
	RejectTrade ActionKind = "REJECT_TRADE"
	UseCard     ActionKind = "USE_CARD"
	EndTurn     ActionKind = "END_TURN"
)

type Action struct {
	Kind    ActionKind     `json:"kind"`
	TileID  int            `json:"tile_id,omitempty"`
	Target  int64          `json:"target_team_id,omitempty"`
	Offer   map[string]any `json:"offer,omitempty"`
}

type Event struct {
	Seq    int64
	TeamID int64
	Action ActionKind
	Kind   string // Buy|Rent|Build|Trade|Tax|Move|Card|Jail
	Detail map[string]any
	Round  int
}
